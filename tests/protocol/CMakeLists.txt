cmake_minimum_required(VERSION 3.30)
project(NkMailBackendProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
if(NOT TARGET mailio)
    set(MAILIO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(MAILIO_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    add_subdirectory(../../third_party/mailio mailio)
endif()
add_executable(backend-probe probe.cpp ../../src/mail/backend.cpp)
target_link_libraries(backend-probe PRIVATE Qt6::Core Qt6::Gui mailio)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME loopback-protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/fixture.py $<TARGET_FILE:backend-probe>)
set_tests_properties(loopback-protocol PROPERTIES TIMEOUT 120 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
